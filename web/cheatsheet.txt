# Сlean Project
rm -rf node_modules package-lock.json   
Remove-Item -Recurse -Force node_modules
npm cache clean --force    

# Build & Run Prod 
# http://localhost:8080/MangaOfflineViewer/
ng build 
ng build --configuration production-local
http-server dist/MangaOfflineViewerAngular/MangaOfflineViewer -p 8080

# GitHub Page
ng build --configuration production-gh
ng deploy 
ng deploy --base-href=/MangaOfflineViewer/

# Run Dev
ng serve --hmr=false 
ng serve --open 

--open - open browswer
--hmr - Hot Module Replacement - Hot Reload
