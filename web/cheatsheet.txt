# Сlean Project
rm -rf node_modules package-lock.json   
Remove-Item -Force package-lock.json    
Remove-Item -Recurse -Force node_modules
Remove-Item -Recurse -Force .angular
Remove-Item -Recurse -Force node_modules\.cache
npm cache clean --force    

# Build & Run Prod 
# http://localhost:8080/MangaOfflineViewer/
npm run version:generate && npx ng build
npm run build
http-server dist/MangaOfflineViewerAngular/MangaOfflineViewer -p 8080

# GitHub Page
npm run build:gh
--ng deploy 
--ng deploy --base-href=/MangaOfflineViewer/
npx angular-cli-ghpages --dir=dist/MangaOfflineViewerAngular/MangaOfflineViewer
OR npm run deploy:gh

# Run Dev
npm start -- --hmr=false
npm run start:open

--open - open browswer
--hmr - Hot Module Replacement - Hot Reload
